<button id="permission-btn">Tap to Start Motion</button>  

<canvas id="foldCanvas"></canvas>  



<script>  

    const canvas = document.getElementById('foldCanvas');  

    const ctx = canvas.getContext('2d');  

    const btn = document.getElementById('permission-btn');  



    const homeScreenImg = new Image();  

    homeScreenImg.src = 'homescreen.png'; // Your iPad home screen screenshot  



    let tiltAngle = 0;  

    let foldFactor = 0;  



    // --- IPAD APP HOTSPOTS (X, Y percentages relative to screen width/height) ---  

    // Coordinates map to standard iPad Landscape positions  

    const appHotspots = [  

        // Row 1  

        { name: 'Roblox', x: 0.18, y: 0.22, radius: 0.06, url: 'roblox://' },  

        { name: 'YouTube', x: 0.38, y: 0.22, radius: 0.06, url: 'youtube://' },  

        { name: 'Discord', x: 0.58, y: 0.22, radius: 0.06, url: 'discord://' },  

        { name: 'Snapchat', x: 0.78, y: 0.22, radius: 0.06, url: 'snapchat://' },  



        // Dock (Bottom)  

        { name: 'Messages', x: 0.40, y: 0.90, radius: 0.05, url: 'sms://' },  

        { name: 'Safari', x: 0.50, y: 0.90, radius: 0.05, url: 'https://apple.com' }  

    ];  



    function resize() {  

        // Native pixel ratio scaling for iPad Retina display  

        canvas.width = window.innerWidth * window.devicePixelRatio;  

        canvas.height = window.innerHeight * window.devicePixelRatio;  

    }  

    window.addEventListener('resize', resize);  

    resize();  



    function render() {  

        ctx.clearRect(0, 0, canvas.width, canvas.height);  



        if (homeScreenImg.complete && homeScreenImg.naturalWidth !== 0) {  

            const w = canvas.width;  

            const h = canvas.height;  



            // Calculate squeeze/fold along the center axis  

            foldFactor = Math.sin(tiltAngle * Math.PI / 180);  

            foldFactor = Math.max(0, Math.min(1, foldFactor));  



            const currentWidth = w * (1 - foldFactor * 0.70);  



            ctx.save();  

            ctx.filter = `blur(${foldFactor * 16}px)`;  

            ctx.drawImage(homeScreenImg, 0, 0, w, h, 0, 0, currentWidth, h);  

            ctx.restore();  



            // Realistic hinge shadow gradient along the edge  

            if (foldFactor > 0.02) {  

                const gradient = ctx.createLinearGradient(currentWidth - 60, 0, currentWidth, 0);  

                gradient.addColorStop(0, 'rgba(0,0,0,0)');  

                gradient.addColorStop(1, `rgba(0,0,0,${foldFactor * 0.85})`);  

                ctx.fillStyle = gradient;  

                ctx.fillRect(currentWidth - 60, 0, 60, h);  

            }  

        }  



        requestAnimationFrame(render);  

    }  



    // Tap Detection  

    canvas.addEventListener('pointerdown', (e) => {  

        if (foldFactor > 0.25) return; // Prevent launch if screen is mid-fold  



        const rect = canvas.getBoundingClientRect();  

        const tapX = (e.clientX - rect.left) / rect.width;  

        const tapY = (e.clientY - rect.top) / rect.height;  



        appHotspots.forEach(app => {  

            const distance = Math.hypot(tapX - app.x, tapY - app.y);  

            if (distance < app.radius) {  

                window.location.href = app.url;  

            }  

        });  

    });  



    // Request Gyroscope Motion Permissions (iPadOS Requirement)  

    btn.addEventListener('click', () => {  

        if (typeof DeviceOrientationEvent !== 'undefined' && typeof DeviceOrientationEvent.requestPermission === 'function') {  

            DeviceOrientationEvent.requestPermission()  

                .then(response => {  

                    if (response === 'granted') {  

                        window.addEventListener('deviceorientation', handleMotion);  

                        btn.style.display = 'none';  

                    }  

                })  

                .catch(console.error);  

        } else {  

            window.addEventListener('deviceorientation', handleMotion);  

            btn.style.display = 'none';  

        }  

    });  



    function handleMotion(event) {  

        // In Landscape, beta tracks the side-to-side folding rotation  

        const isLandscape = window.innerWidth > window.innerHeight;  

        const rawAngle = isLandscape ? event.beta : event.gamma;  



        if (rawAngle !== null) {  

            tiltAngle = Math.abs(rawAngle);  

        }  

    }  



    render();  

</script>
